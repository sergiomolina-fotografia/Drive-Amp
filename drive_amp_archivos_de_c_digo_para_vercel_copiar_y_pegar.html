⚡ api/stream.js Carpeta /api/

Streaming continuo de audio con bypass de cookies de confirmación y soporte de pausas/saltos (Range):

export default async function handler(req, res) {
  if (req.method === 'OPTIONS') {
    res.setHeader('Access-Control-Allow-Origin', '*');
    res.setHeader('Access-Control-Allow-Methods', 'GET, OPTIONS');
    res.setHeader('Access-Control-Allow-Headers', 'Range, Content-Type');
    return res.status(200).end();
  }

  const { id } = req.query;
  if (!id || typeof id !== 'string') {
    return res.status(400).json({ error: 'Falta el parámetro ?id de Google Drive' });
  }

  const driveUrl = `https://drive.usercontent.google.com/download?id=${encodeURIComponent(id)}&export=download&authuser=0&confirm=t`;

  try {
    const fetchHeaders = {};
    if (req.headers.range) {
      fetchHeaders['Range'] = req.headers.range;
    }

    const driveRes = await fetch(driveUrl, {
      headers: fetchHeaders,
      redirect: 'follow',
    });

    if (!driveRes.ok && driveRes.status !== 206) {
      const altUrl = `https://drive.google.com/uc?export=download&id=${encodeURIComponent(id)}&confirm=t`;
      const altRes = await fetch(altUrl, { headers: fetchHeaders, redirect: 'follow' });
      if (!altRes.ok && altRes.status !== 206) {
        return res.status(driveRes.status).json({
          error: 'No se pudo obtener el audio de Google Drive. Verifica que sea público.'
        });
      }
      return pipeStream(altRes, req, res);
    }

    return pipeStream(driveRes, req, res);
  } catch (err) {
    console.error('Error en stream.js:', err);
    return res.status(500).json({ error: err.message });
  }
}

function pipeStream(sourceRes, req, res) {
  res.setHeader('Access-Control-Allow-Origin', '*');
  res.setHeader('Accept-Ranges', 'bytes');
  res.setHeader('Cache-Control', 'public, max-age=86400');

  const contentType = sourceRes.headers.get('content-type') || 'audio/mpeg';
  res.setHeader('Content-Type', contentType);

  const contentLength = sourceRes.headers.get('content-length');
  if (contentLength) res.setHeader('Content-Length', contentLength);

  const contentRange = sourceRes.headers.get('content-range');
  if (contentRange) {
    res.setHeader('Content-Range', contentRange);
    res.status(206);
  } else {
    res.status(sourceRes.status || 200);
  }

  if (sourceRes.body) {
    const reader = sourceRes.body.getReader();
    const pump = async () => {
      try {
        while (true) {
          const { done, value } = await reader.read();
          if (done) { res.end(); break; }
          if (!res.write(value)) {
            await new Promise(r => res.once('drain', r));
          }
        }
      } catch (err) {
        console.error('Error streaming chunk:', err);
        res.end();
      }
    };
    pump();
  } else {
    res.end();
  }
}
⚡ api/resolve.js Carpeta /api/

Lee y extrae los nombres e IDs de todas las canciones reales de cualquier carpeta pública de Google Drive:

export default async function handler(req, res) {
  if (req.method === 'OPTIONS') {
    res.setHeader('Access-Control-Allow-Origin', '*');
    res.setHeader('Access-Control-Allow-Methods', 'GET, POST, OPTIONS');
    res.setHeader('Access-Control-Allow-Headers', 'Content-Type');
    return res.status(200).end();
  }

  res.setHeader('Access-Control-Allow-Origin', '*');

  let folderId = req.query.id || (req.body && req.body.id);
  const url = req.query.url || (req.body && req.body.url);

  if (!folderId && url) {
    const m = url.match(/folders\/([a-zA-Z0-9_-]+)/) || url.match(/[?&]id=([a-zA-Z0-9_-]+)/);
    if (m) folderId = m[1];
  }

  const fileMatch = url && (url.match(/\/file\/d\/([a-zA-Z0-9_-]+)/) || url.match(/[?&]id=([a-zA-Z0-9_-]+)/));
  if (fileMatch && (!url.includes('/folders/'))) {
    return res.status(200).json({
      type: 'file',
      tracks: [{
        id: fileMatch[1],
        name: 'Pista de Google Drive (' + fileMatch[1].substring(0, 6) + '...)',
        streamUrl: `/api/stream?id=${fileMatch[1]}`
      }]
    });
  }

  if (!folderId) {
    return res.status(400).json({ error: 'No se pudo identificar el ID de carpeta o archivo.' });
  }

  try {
    const embedUrl = `https://drive.google.com/embeddedfolderview?id=${encodeURIComponent(folderId)}#list`;
    const response = await fetch(embedUrl, {
      headers: {
        'User-Agent': 'Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36'
      }
    });

    if (!response.ok) {
      return res.status(404).json({ error: 'Carpeta no accesible o privada. Debe estar compartida como pública.' });
    }

    const html = await response.text();
    const tracks = [];
    const seenIds = new Set();

    const entryRegex = /id="entry-([a-zA-Z0-9_-]+)"[\s\S]*?
([^<]+)<\/div>/g;
    let match;
    while ((match = entryRegex.exec(html)) !== null) {
      const id = match[1];
      const name = match[2].trim();
      if (!seenIds.has(id) && isAudio(name)) {
        seenIds.add(id);
        tracks.push({ id, name, streamUrl: `/api/stream?id=${id}` });
      }
    }

    if (tracks.length === 0) {
      const jsonItems = html.matchAll(/\["([a-zA-Z0-9_-]{25,})","([^"]+\.(mp3|wav|flac|ogg|m4a|aac))"/gi);
      for (const m of jsonItems) {
        const id = m[1];
        const name = m[2];
        if (!seenIds.has(id)) {
          seenIds.add(id);
          tracks.push({ id, name, streamUrl: `/api/stream?id=${id}` });
        }
      }
    }

    if (tracks.length === 0) {
      const fallbackMatches = html.matchAll(/\/file\/d\/([a-zA-Z0-9_-]+)[^>]*>([^<]+)<\/a>/g);
      for (const m of fallbackMatches) {
        const id = m[1];
        const name = m[2].trim();
        if (!seenIds.has(id)) {
          seenIds.add(id);
          tracks.push({ id, name, streamUrl: `/api/stream?id=${id}` });
        }
      }
    }

    if (tracks.length === 0) {
      return res.status(404).json({
        error: 'No se encontraron pistas de audio en la carpeta. Verifica que sean públicas.'
      });
    }

    return res.status(200).json({
      type: 'folder',
      folderId,
      count: tracks.length,
      tracks
    });

  } catch (err) {
    console.error('Error en resolve.js:', err);
    return res.status(500).json({ error: 'Error del servidor: ' + err.message });
  }
}

function isAudio(filename) {
  return /\.(mp3|wav|flac|ogg|m4a|aac|opus|webm|wma)$/i.test(filename);
}
